type clockInput struct {
	Year, Month, Day, Hour, Minute, Second, Nanosecond, OffsetSeconds int
}

func (c clockInput) value() time.Time {
	return time.Date(c.Year, time.Month(c.Month), c.Day, c.Hour, c.Minute, c.Second, c.Nanosecond, time.FixedZone("synthetic", c.OffsetSeconds))
}

var fixed = clockInput{2026, 8, 8, 12, 21, 18, 123456789, 8 * 60 * 60}

type eventInput struct {
	Module      string            `json:"module"`
	Kind        string            `json:"kind"`
	Operation   string            `json:"operation"`
	Resource    string            `json:"resource"`
	Route       string            `json:"route"`
	Target      string            `json:"target"`
	Proxy       string            `json:"proxy"`
	UserAgent   string            `json:"user_agent"`
	Reason      string            `json:"reason"`
	Status      int               `json:"status"`
	Count       int               `json:"count"`
	RequestID   uint64            `json:"request_id"`
	DurationNS  int64             `json:"duration_ns"`
	StringBytes map[string]string `json:"go_string_bytes_base64,omitempty"`
}

func input(e core.Event) eventInput {
	v := eventInput{Module: string(e.Module), Kind: string(e.Kind), Operation: e.Operation, Resource: e.Resource, Route: e.Route, Target: e.Target, Proxy: e.Proxy, UserAgent: e.UserAgent, Reason: string(e.Reason), Status: e.Status, Count: e.Count, RequestID: e.RequestID, DurationNS: int64(e.Duration)}
	for name, value := range map[string]string{"module": v.Module, "kind": v.Kind, "operation": v.Operation, "resource": v.Resource, "route": v.Route, "target": v.Target, "proxy": v.Proxy, "user_agent": v.UserAgent, "reason": v.Reason} {
		if !utf8.ValidString(value) {
			if v.StringBytes == nil {
				v.StringBytes = map[string]string{}
			}
			v.StringBytes[name] = base64.StdEncoding.EncodeToString([]byte(value))
		}
	}
	return v
}

type output struct {
	Bytes string `json:"output_base64"`
	UTF8  bool   `json:"output_is_utf8"`
	Text  string `json:"output_text,omitempty"`
}

func observed(data []byte) output {
	v := output{Bytes: base64.StdEncoding.EncodeToString(data), UTF8: utf8.Valid(data)}
	if v.UTF8 {
		v.Text = string(data)
	}
	return v
}

func errText(err error) string {
	if err == nil {
		return ""
	}
	return err.Error()
}

type renderRow struct {
	Name          string     `json:"name"`
	Compatibility string     `json:"compatibility"`
	Constructor   string     `json:"constructor"`
	Format        string     `json:"format"`
	Clock         clockInput `json:"clock"`
	Event         eventInput `json:"event"`
	Output        output     `json:"output"`
	ClockCalls    int        `json:"clock_calls"`
	Error         string     `json:"error"`
	NilWriter     bool       `json:"nil_writer,omitempty"`
}

func render(name string, event core.Event, clock clockInput, constructor, format, compatibility string) renderRow {
	var sink bytes.Buffer
	calls := 0
	now := func() time.Time { calls++; return clock.value() }
	var p *cli.Presenter
	if constructor == "clock" {
		p = cli.NewPresenterWithClock(&sink, now)
	} else {
		p = cli.NewPresenterWithFormat(&sink, format, now)
	}
	p.Emit(event)
	return renderRow{name, compatibility, constructor, format, clock, input(event), observed(sink.Bytes()), calls, errText(p.Err()), false}
}

type writerStep struct {
	Count string `json:"count"`
	Error string `json:"error"`
}

type writeCall struct {
	Method    string `json:"method"`
	Record    output `json:"record"`
	ReturnedN int    `json:"returned_n"`
	Error     string `json:"error"`
}

type errorState struct {
	Error   string `json:"error"`
	IsFirst bool   `json:"is_first_error_identity"`
	IsLater bool   `json:"is_later_error_identity"`
}

type writerOnly struct {
	steps        []writerStep
	calls        []writeCall
	stored       bytes.Buffer
	first, later error
}

func (w *writerOnly) record(data []byte, method string) (int, error) {
	step := writerStep{"all", ""}
	if len(w.calls) < len(w.steps) {
		step = w.steps[len(w.calls)]
	}
	n := len(data)
	switch step.Count {
	case "zero":
		n = 0
	case "short":
		n = len(data) - 1
	case "negative":
		n = -1
	case "over":
		n = len(data) + 1
	}
	var err error
	switch step.Error {
	case "first":
		err = w.first
	case "later":
		err = w.later
	}
	w.calls = append(w.calls, writeCall{method, observed(data), n, errText(err)})
	if n > 0 && n <= len(data) {
		_, _ = w.stored.Write(data[:n])
	}
	return n, err
}

func (w *writerOnly) Write(data []byte) (int, error) { return w.record(data, "Write") }

type stringWriter struct{ *writerOnly }

func (w stringWriter) WriteString(value string) (int, error) {
	return w.record([]byte(value), "WriteString")
}

type writerRow struct {

	AfterEach              []errorState `json:"error_after_each_emit"`

func writerCases() []writerRow {
	var rows []writerRow
	for _, spec := range []struct {
		name, format  string
		stringWriter  bool
		steps         []writerStep
		compatibility string
	}{
		{"first_error_preserved_write", "text", false, []writerStep{{"zero", "first"}, {"short", "later"}, {"all", ""}}, "shared"},
		{"first_error_preserved_stringwriter", "json", true, []writerStep{{"short", "first"}, {"zero", "later"}, {"all", ""}}, "shared"},
		{"short_nil_error_write", "text", false, []writerStep{{"zero", ""}, {"short", ""}, {"all", ""}}, "shared"},
		{"short_nil_error_stringwriter", "json", true, []writerStep{{"short", ""}, {"zero", ""}, {"all", ""}}, "shared"},
		{"success_before_first_error", "json", false, []writerStep{{"all", ""}, {"all", "first"}, {"all", "later"}}, "shared"},
		{"invalid_writer_counts", "text", false, []writerStep{{"negative", ""}, {"over", ""}, {"all", ""}}, "go_only_writer_contract_violation"},
	} {
		w := &writerOnly{steps: spec.steps, calls: []writeCall{}, first: errors.New("synthetic first writer error"), later: errors.New("synthetic later writer error")}
		var sink io.Writer = w
		if spec.stringWriter {
			sink = stringWriter{w}
		}
		row := writerRow{Name: spec.name, Compatibility: spec.compatibility, Format: spec.format, Clock: fixed, ImplementsStringWriter: spec.stringWriter, Steps: spec.steps, Events: []eventInput{}, AfterEach: []errorState{}}
		p := cli.NewPresenterWithFormat(sink, spec.format, func() time.Time { row.ClockCalls++; return fixed.value() })
		for _, kind := range []core.EventKind{core.EventStarted, core.EventCompleted, core.EventFailed} {
			e := core.Event{Module: core.ModulePixivCLI, Kind: kind, Operation: "synthetic command", Reason: core.ReasonCommandFailed}
			row.Events = append(row.Events, input(e))
			p.Emit(e)
			err := p.Err()
			row.AfterEach = append(row.AfterEach, errorState{errText(err), err == w.first, err == w.later})
		}
		row.Calls, row.Stored = w.calls, observed(w.stored.Bytes())
		rows = append(rows, row)
	}
	return rows
}

type defaultRow struct {
	Name              string     `json:"name"`
	Constructor       string     `json:"constructor"`
	Format            string     `json:"format"`
	NilWriter         bool       `json:"nil_writer"`
	NilClock          bool       `json:"nil_clock"`
	Event             eventInput `json:"event"`
	NormalizedOutput  string     `json:"normalized_output"`
	ClockInCallWindow bool       `json:"clock_in_call_window"`
	Error             string     `json:"error"`
}

func defaults() []defaultRow {
	var rows []defaultRow
	event := core.Event{Module: core.ModulePixivCLI, Kind: core.EventStarted, Operation: "work"}
	for _, spec := range []struct {
		name, constructor, format string
		nilWriter                 bool
	}{
		{"default_constructor", "default", "text", false},
		{"nil_clock_constructor", "clock", "text", false},
		{"nil_clock_text_format", "format", "text", false},
		{"nil_clock_json_format", "format", "json", false},
		{"nil_writer_default_constructor", "default", "text", true},
		{"nil_writer_clock_constructor", "clock", "text", true},
		{"nil_writer_json_format", "format", "json", true},
	} {
		var b bytes.Buffer
		var sink io.Writer = &b
		if spec.nilWriter {
			sink = nil
		}
		var p *cli.Presenter
		switch spec.constructor {
		case "default":
			p = cli.NewPresenter(sink)
		case "clock":
			p = cli.NewPresenterWithClock(sink, nil)
		default:
			p = cli.NewPresenterWithFormat(sink, spec.format, nil)
		}
		before := time.Now()
		p.Emit(event)
		after := time.Now()
		row := defaultRow{Name: spec.name, Constructor: spec.constructor, Format: spec.format, NilWriter: spec.nilWriter, NilClock: true, Event: input(event), Error: errText(p.Err()), ClockInCallWindow: true}
		if !spec.nilWriter && spec.format == "json" {
			var record map[string]any
			if err := json.Unmarshal(b.Bytes(), &record); err != nil {
				panic(err)
			}
			stamp := record["time"].(string)
			parsed, err := time.Parse(time.RFC3339Nano, stamp)
			row.ClockInCallWindow = err == nil && !parsed.Before(before) && !parsed.After(after)
			row.NormalizedOutput = strings.Replace(b.String(), stamp, "<LOCAL_CLOCK>", 1)
		} else if !spec.nilWriter {
			r := regexp.MustCompile(`\[Pixiv CLI\] ([0-9]{2}:[0-9]{2}:[0-9]{2}) Started work\.\n`)
			match := r.FindStringSubmatch(b.String())
			row.ClockInCallWindow = len(match) == 2 && (match[1] == before.Format("15:04:05") || match[1] == after.Format("15:04:05"))
			if len(match) == 2 {
				row.NormalizedOutput = strings.Replace(b.String(), match[1], "<LOCAL_CLOCK>", 1)
			}
		}
		if !row.ClockInCallWindow {
			panic("default clock is outside observed call window")
		}
		rows = append(rows, row)
	}
	return rows
}

type nilWriter struct{}

func (w *nilWriter) Write(data []byte) (int, error) {
	if w == nil {
		panic("synthetic typed-nil writer")
	}
	return len(data), nil
}

type goOnlyRow struct {
	Name          string `json:"name"`
	Compatibility string `json:"compatibility"`
	Panic         string `json:"panic"`
	Error         string `json:"error"`
	ClockCalls    int    `json:"clock_calls"`
	Writes        int    `json:"writes"`
}

func catch(f func()) (panicText string) {
	defer func() {
		if value := recover(); value != nil {
			panicText = fmt.Sprint(value)
		}
	}()
	f()
	return ""
}

func goOnly() []goOnlyRow {
	var rows []goOnlyRow
	var nilPresenter *cli.Presenter
	rows = append(rows, goOnlyRow{Name: "nil_presenter_receiver", Compatibility: "go_only_nil_receiver", Panic: catch(func() { nilPresenter.Emit(core.Event{}) }), Error: errText(nilPresenter.Err())})
	for _, name := range []string{"typed_nil_writer", "injected_clock_panics"} {
		row := goOnlyRow{Name: name, Compatibility: "go_only_nil_type_or_panic"}
		w := &writerOnly{calls: []writeCall{}}
		var sink io.Writer = w
		if name == "typed_nil_writer" {
			var typedNil *nilWriter
			sink = typedNil
		}
		p := cli.NewPresenterWithFormat(sink, "text", func() time.Time {
			row.ClockCalls++
			if name == "injected_clock_panics" {
				panic("synthetic injected clock panic")
			}
			return fixed.value()
		})
		row.Panic = catch(func() { p.Emit(core.Event{Module: core.ModulePixivCLI, Kind: core.EventStarted}) })
		row.Error, row.Writes = errText(p.Err()), len(w.calls)
		rows = append(rows, row)
	}
	return rows
}

type concurrentWriter struct {
	active, max atomic.Int64
	mu          sync.Mutex
	records     []string
}

func updateMax(max *atomic.Int64, value int64) {
	for old := max.Load(); value > old; old = max.Load() {
		if max.CompareAndSwap(old, value) {
			return
		}
	}
}

func (w *concurrentWriter) Write(data []byte) (int, error) {
	active := w.active.Add(1)
	updateMax(&w.max, active)
	for i := 0; i < 8; i++ {
		runtime.Gosched()
	}
	w.mu.Lock()
	w.records = append(w.records, string(data))
	w.mu.Unlock()
	w.active.Add(-1)
	return len(data), nil
}

func concurrency() []concurrentRow {
	var rows []concurrentRow
	for _, format := range []string{"text", "json"} {
		const count = 32
		w := &concurrentWriter{records: []string{}}
		var clockCalls, active, maximum atomic.Int64
		var barrier, workers sync.WaitGroup
		barrier.Add(count)
		p := cli.NewPresenterWithFormat(w, format, func() time.Time {
			clockCalls.Add(1)
			updateMax(&maximum, active.Add(1))
			barrier.Done()
			barrier.Wait()
			active.Add(-1)
			return fixed.value()
		})
		row := concurrentRow{Name: "concurrent_intact_records_" + format, Format: format, Clock: fixed, Events: []eventInput{}, SortedRecords: []output{}, IntactSingleRecords: true}
		for i := 0; i < count; i++ {
			e := core.Event{Module: core.ModulePixivMCP, Kind: core.EventStarted, Operation: fmt.Sprintf("synthetic-operation-%02d", i), RequestID: uint64(i + 1)}
			row.Events = append(row.Events, input(e))
			workers.Add(1)
			go func() { defer workers.Done(); p.Emit(e) }()
		}
		workers.Wait()
		sort.Strings(w.records)
		for _, record := range w.records {
			row.SortedRecords = append(row.SortedRecords, observed([]byte(record)))
			if strings.Count(record, "\n") != 1 || !strings.HasSuffix(record, "\n") {
				row.IntactSingleRecords = false
			}
		}
		row.ClockCalls, row.MaxConcurrentClockCalls, row.MaxConcurrentWriterCalls, row.Error = clockCalls.Load(), maximum.Load(), w.max.Load(), errText(p.Err())
		if len(w.records) != count || !row.IntactSingleRecords || row.MaxConcurrentWriterCalls != 1 || row.MaxConcurrentClockCalls != count {
			panic("concurrent presenter record contract failed")
		}
		rows = append(rows, row)
	}
	return rows
}
