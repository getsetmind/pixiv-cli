type migrationLoginOwnershipTransport struct {
	calls   atomic.Int32
	closes  atomic.Int32
	inspect func(*http.Request) (*http.Response, error)
}

func (m *migrationLoginOwnershipTransport) RoundTrip(r *http.Request) (*http.Response, error) {
	m.calls.Add(1)
	if m.inspect != nil {
		return m.inspect(r)
	}
	return migrationLoginOwnershipResponse(r), nil
}

func (m *migrationLoginOwnershipTransport) CloseIdleConnections() { m.closes.Add(1) }

func migrationLoginOwnershipResponse(r *http.Request) *http.Response {
	return &http.Response{StatusCode: http.StatusOK, Header: http.Header{"Content-Type": {"application/json"}}, Body: io.NopCloser(strings.NewReader(loginMigrationSuccess)), Request: r}
}

func migrationLoginOwnershipError(err error) map[string]any {
	result := map[string]any{"reason": sdk.ReasonOf(err), "message": "", "canceled": errors.Is(err, context.Canceled), "deadline_exceeded": errors.Is(err, context.DeadlineExceeded)}
	if err != nil {
		result["message"] = err.Error()
	}
	var classified *sdk.Error
	if errors.As(err, &classified) {
		result["operation"] = classified.Operation
		result["detail"] = classified.Detail
		result["transport"] = classified.Transport
		result["http_status"] = classified.HTTPStatus
	}
	return result
}

func migrationLoginOwnershipAwait[T any](t *testing.T, channel <-chan T, label string) T {
	t.Helper()
	select {
	case value := <-channel:
		return value
	case <-time.After(3 * time.Second):
		t.Fatalf("owned login %s did not complete", label)
		var zero T
		return zero
	}
}

func migrationLoginOwnershipBegin(t *testing.T, client *http.Client) *LoginSession {
	t.Helper()

	}
	return session
}

func migrationLoginOwnershipPKCE(session *LoginSession, request *http.Request) map[string]any {
	authorization, _ := url.Parse(session.AuthorizationURL())
	_ = request.ParseForm()
	sum := sha256.Sum256([]byte(request.PostForm.Get("code_verifier")))
	return map[string]any{"method": request.Method, "url": request.URL.String(), "code": request.PostForm.Get("code"), "pkce_matches_authorization": base64.RawURLEncoding.EncodeToString(sum[:]) == authorization.Query().Get("code_challenge")}
}

type migrationLoginOwnershipRandom struct {
	reads []int
	step  func(int)
}

func (r *migrationLoginOwnershipRandom) Read(p []byte) (int, error) {
	r.reads = append(r.reads, len(p))
	if r.step != nil {
		r.step(len(r.reads))
	}
	for i := range p {
		p[i] = byte(i + len(r.reads))
	}
	return len(p), nil
}

func migrationLoginOwnershipConstruction(t *testing.T, mode string) migrationLoginOwnershipRow {
	oldRandom, oldTransport := rand.Reader, http.DefaultTransport
	defer func() { rand.Reader, http.DefaultTransport = oldRandom, oldTransport }()
	reader := &migrationLoginOwnershipRandom{}
	rand.Reader = reader
	original, replacement := &http.Transport{MaxIdleConns: 7}, &http.Transport{MaxIdleConns: 19}
	http.DefaultTransport = original
	reader.step = func(step int) {
		if step == 2 {
			switch mode {
			case "client_selection_after_randomness":
				http.DefaultTransport = replacement
			case "nil_default_transport":
				http.DefaultTransport = nil
			case "incompatible_default_transport":
				http.DefaultTransport = &migrationLoginOwnershipTransport{}
			}
		}
	}
	var session *LoginSession
	var returned error
	panicked := false
	func() {
		defer func() { panicked = recover() != nil }()
		options := LoginOptions{}
		if mode == "injected_bypasses_nil_default" {
			http.DefaultTransport = nil
			options.HTTPClient = &http.Client{Transport: &migrationLoginOwnershipTransport{}}
		}
		session, returned = BeginLogin(options)
	}()
	observation := map[string]any{"random_read_sizes": reader.reads, "go_only_panicked": panicked, "returned_session_nil": session == nil, "error": migrationLoginOwnershipError(returned)}
	if session != nil {
		observation["go_only_owned"] = session.state.selfHTTP
		if tr, ok := session.state.httpClient.Transport.(*http.Transport); ok {
			observation["go_only_cloned_transport_distinct"] = tr != http.DefaultTransport
			observation["captured_max_idle_connections"] = tr.MaxIdleConns
		}
		session.CloseIdleConnections()
	}
	return migrationLoginOwnershipRow{"go_only/construction/" + mode, map[string]any{"entry": "public BeginLogin; isolated process overrides crypto/rand.Reader and default RoundTripper", "mode": mode, "randomness": "deterministic synthetic bytes; no production randomness change"}, observation}
}

			t.Error(err)
		}

			active <- struct{}{}

func TestMigrationLoginOwnershipChild(t *testing.T) {

		row = migrationLoginOwnershipConstruction(t, mode)
	}
	encoded, err := json.Marshal(row)
	if err != nil {
		t.Fatal(err)
	}
	fmt.Println(migrationLoginOwnershipPrefix + string(encoded))
}

type migrationLoginOwnershipFailureRandom struct{ stage, reads int }

func (r *migrationLoginOwnershipFailureRandom) Read(p []byte) (int, error) {
	r.reads++
	fmt.Printf("OWNED_LOGIN_ENTROPY_READ:%d\n", len(p))
	if r.reads == r.stage {
		return 0, errors.New("owned synthetic entropy unavailable")
	}
	for i := range p {
		p[i] = byte(i)
	}
	return len(p), nil
}

		var reads []int
		fatal := false
		for line := range strings.SplitSeq(string(output), "\n") {
			if strings.HasPrefix(line, "OWNED_LOGIN_ENTROPY_READ:") {
				n, _ := strconv.Atoi(strings.TrimPrefix(line, "OWNED_LOGIN_ENTROPY_READ:"))
				reads = append(reads, n)
			}
			if strings.Contains(line, "fatal error: crypto/rand: failed to read random data") {
				fatal = true
			}
		}
		if !fatal {
			t.Fatalf("unexpected entropy child failure: %v\n%s", err, output)
		}
		return migrationLoginOwnershipRow{"go_only/construction/" + mode, map[string]any{"entry": "public BeginLogin; isolated process overrides crypto/rand.Reader with a synthetic read error", "mode": mode}, map[string]any{"random_read_sizes": reads, "go_only_process_failed": true, "go_only_official_rand_read_fatal": fatal, "recoverable_begin_login_error_observed": false}}
	}
	if err != nil {
		t.Fatalf("ownership child %s: %v\n%s", mode, err, output)
	}
	for line := range strings.SplitSeq(string(output), "\n") {
		if strings.HasPrefix(line, migrationLoginOwnershipPrefix) {
			var row migrationLoginOwnershipRow
			if err := json.Unmarshal([]byte(strings.TrimPrefix(line, migrationLoginOwnershipPrefix)), &row); err != nil {
				t.Fatal(err)
			}
			return row
		}
	}
	t.Fatalf("ownership child %s did not report its observation: %s", mode, output)
	return migrationLoginOwnershipRow{}
}
