use pixiv_sdk::{
    Client,
    resource::{OpenResourceRequest, ResourceRef},
    transport::HttpTransport,
};
use serde_json::{Value, json};
use std::{
    fs,
    io::{BufRead, BufReader, Read, Write},
    path::{Path, PathBuf},
    process::{Child, Command, Stdio},
    sync::{
        Arc,
        atomic::{AtomicUsize, Ordering},
        mpsc,
    },
    thread::JoinHandle,
    time::{Duration, Instant},
};
use tokio::{io::AsyncReadExt, task::JoinHandle as Task};

const BOUND: Duration = Duration::from_secs(10);
const API_HOST: &str = "app-api.pixiv.net";
const MEDIA_HOST: &str = "i.pximg.net";
const MEDIA_TARGET: &str = "/idle-native-h1.fixture";
static DIRECTORY_SEQUENCE: AtomicUsize = AtomicUsize::new(0);

type NativeClient = Client<HttpTransport>;

struct Peer {
    child: Child,
    events: mpsc::Receiver<Value>,
    reader: Option<JoinHandle<()>>,
    pending: Vec<Value>,
    observed: Vec<Value>,
    directory: PathBuf,
    proxy: String,
    name: String,
}

impl Peer {
    fn new(name: &str) -> Self {
        let root = Path::new(env!("CARGO_MANIFEST_DIR")).join("../..");
        let directory = std::env::temp_dir().join(format!(
            "pixiv-owned-ordinary-sdk-h1-{}-{}",
            std::process::id(),
            DIRECTORY_SEQUENCE.fetch_add(1, Ordering::SeqCst)
        ));
        fs::create_dir(&directory).unwrap();
        let go = std::env::var_os("PIXIV_NATIVE_GO")
            .map(PathBuf::from)
            .unwrap_or_else(|| root.join("../go-toolchain/go/bin/go"));
        let version = Command::new(&go).arg("version").output().unwrap();
        assert!(version.status.success());
        assert!(String::from_utf8_lossy(&version.stdout).starts_with("go version go1.27.1 "));
        let helper = directory.join("owned-ordinary-h1-peer");
        let mut build = Command::new(&go)
            .current_dir(&root)
            .args(["build", "-mod=readonly", "-o"])
            .arg(&helper)
            .arg("crates/pixiv-sdk/tests/helpers/ordinary_h1_peer.go")
            .env("GOPROXY", "off")
            .env("GOWORK", "off")
            .env("GOTOOLCHAIN", "local")
            .stdout(Stdio::null())
            .stderr(Stdio::piped())
            .spawn()
            .unwrap();
        let mut stderr = build.stderr.take().unwrap();
        let (send, output) = mpsc::channel();
        let build_reader = std::thread::spawn(move || {
            let mut bytes = Vec::new();
            stderr.read_to_end(&mut bytes).unwrap();
            let _ = send.send(bytes);
        });
        let output = output.recv_timeout(Duration::from_secs(60));
        if output.is_err() {
            let _ = build.kill();
            let _ = build.wait();
        }
        build_reader.join().unwrap();
        let output = output.expect("bounded owned Go helper compilation");
        assert!(
            build.wait().unwrap().success(),
            "{}",
            String::from_utf8_lossy(&output)
        );
        let mut child = Command::new(helper)
            .arg(&directory)
            .stdin(Stdio::piped())
            .stdout(Stdio::piped())
            .spawn()
            .unwrap();
        let stdout = child.stdout.take().unwrap();
        let (send, events) = mpsc::channel();
        let reader = std::thread::spawn(move || {
            for line in BufReader::new(stdout).lines() {
                let value = serde_json::from_str(&line.unwrap()).unwrap();
                if send.send(value).is_err() {
                    break;
                }
            }
        });
        let mut peer = Self {
            child,
            events,
            reader: Some(reader),
            pending: vec![],
            observed: vec![],
            directory,
            proxy: String::new(),
            name: name.into(),
        };
        let ready = peer.wait(
            |event| event["event"] == "ready",
            "TLS CONNECT peer readiness",
        );
        peer.proxy = ready["proxy"].as_str().unwrap().into();
        peer
    }

    fn command(&mut self, command: Value) {
        let stdin = self.child.stdin.as_mut().unwrap();
        serde_json::to_writer(&mut *stdin, &command).unwrap();
        writeln!(stdin).unwrap();
        stdin.flush().unwrap();
    }

    fn wait(&mut self, matches: impl Fn(&Value) -> bool, what: &str) -> Value {
        if let Some(index) = self.pending.iter().position(&matches) {
            return self.pending.remove(index);
        }
        let until = Instant::now() + BOUND;
        loop {
            let value = self
                .events
                .recv_timeout(until.saturating_duration_since(Instant::now()))
                .unwrap_or_else(|error| {
                    panic!("{}: bounded peer barrier {what}: {error}", self.name)
                });
            assert_ne!(value["event"], "error", "owned peer failure: {value}");
            self.observed.push(value.clone());
            if matches(&value) {
                return value;
            }
            self.pending.push(value);
        }
    }

    fn plan(&mut self, name: &str, id: i64, phase: &str, resource: bool) {
        let (host, target) = if resource {
            (MEDIA_HOST, MEDIA_TARGET.to_owned())
        } else {
            (API_HOST, format!("/v1/illust/detail?illust_id={id}"))
        };
        self.command(json!({"command":"plan", "name":name, "id":id, "phase":phase, "host":host, "target":target}));
        self.wait(
            |event| event["event"] == "planned" && event["name"] == name,
            "source plan installation",
        );
    }

    fn arrived(&mut self, name: &str, phase: &str, connection: u64) {
        let request = self.wait(
            |event| event["event"] == "request" && event["wire"]["step"] == name,
            "real peer request",
        );
        assert_eq!(request["wire"]["connection"], connection);
        assert_eq!(request["wire"]["method"], "GET");
        assert_eq!(request["wire"]["protocol"], "HTTP/1.1");
        if !phase.is_empty() {
            self.wait(
                |event| {
                    event["event"] == "barrier" && event["name"] == name && event["phase"] == phase
                },
                "source-driven withheld response",
            );
        }
    }

    fn release(&mut self, name: &str) {
        self.command(json!({"command":"release", "name":name}));
    }

    fn closed(&mut self, connection: u64) {
        let event = self.wait(
            |event| event["event"] == "peer_closed" && event["connection"] == connection,
            "independent TLS peer EOF/reset observation",
        );
        assert_eq!(event["owned_cleanup"], false);
    }

    fn transport(&self) -> HttpTransport {
        let root =
            reqwest::Certificate::from_pem(&fs::read(self.directory.join("root.pem")).unwrap())
                .unwrap();
        let client = reqwest::Client::builder()
            .no_proxy()
            .proxy(reqwest::Proxy::all(&self.proxy).unwrap())
            .tls_certs_only([root])
            .http1_only()
            .redirect(reqwest::redirect::Policy::none())
            .timeout(BOUND)
            .pool_idle_timeout(Duration::from_secs(90))
            .pool_idle_retention(true)
            .pool_max_idle_per_host(2)
            .pool_max_idle_connections(100)
            .build()
            .unwrap();
        HttpTransport::from_client(client)
    }

    fn owned_client(&self) -> Arc<NativeClient> {
        Arc::new(Client::with_owned_transport("fixture-access", self.transport()).unwrap())
    }

    fn snapshot(&mut self) -> Value {
        self.command(json!({"command":"snapshot"}));
        self.wait(|event| event["event"] == "snapshot", "peer snapshot")
    }

    fn finish(mut self, witness: Value) {
        self.command(json!({"command":"stop"}));
        self.wait(|event| event["event"] == "stopped", "owned peer stop");
        assert!(self.child.wait().unwrap().success());
        if let Some(directory) = std::env::var_os("PIXIV_ORDINARY_SDK_H1_WITNESS_DIR") {
            let directory = PathBuf::from(directory);
            fs::create_dir_all(&directory).unwrap();
            fs::write(
                directory.join(format!("{}.json", self.name)),
                serde_json::to_vec_pretty(
                    &json!({"portable_observation":witness, "raw_peer_events":self.observed}),
                )
                .unwrap(),
            )
            .unwrap();
        }
    }
}

impl Drop for Peer {
    fn drop(&mut self) {
        if self.child.try_wait().ok().flatten().is_none() {
            let _ = self.child.kill();
            let _ = self.child.wait();
        }
        if let Some(reader) = self.reader.take() {
            reader.join().unwrap();
        }
        let _ = fs::remove_dir_all(&self.directory);
    }
}

fn artwork(client: &Arc<NativeClient>, id: i64) -> Task<i64> {
    let client = Arc::clone(client);
    tokio::spawn(async move { client.artwork(id).await.unwrap().id })
}

async fn finish_artwork(task: Task<i64>, step: &str, id: i64) -> Value {
    assert_eq!(
        tokio::time::timeout(BOUND, task).await.unwrap().unwrap(),
        id
    );
    json!({"step":step, "operation":"Artwork", "success":true, "id":id})
}

async fn complete_artwork(
    peer: &mut Peer,
    client: &Arc<NativeClient>,
    step: &str,
    id: i64,
    connection: u64,
) -> Value {
    peer.plan(step, id, "", false);
    let task = artwork(client, id);
    peer.arrived(step, "", connection);
    finish_artwork(task, step, id).await
}

fn fixture(name: &str) -> Value {
    let rows: Vec<Value> =
        serde_json::from_str(include_str!("fixtures/client-idle-native-h1.json")).unwrap();
    rows.into_iter().find(|row| row["name"] == name).unwrap()
}

fn compare_native(
    peer: &mut Peer,
    name: &str,
    public: Vec<Value>,
    close_idle_calls: u64,
    closures: Vec<Value>,
) -> Value {
    let expected = fixture(name);
    let snapshot = peer.snapshot();
    let actual_requests: Vec<Value> = snapshot["requests"]
        .as_array()
        .unwrap()
        .iter()
        .map(|wire| {
            json!({"step":wire["step"], "connection":wire["connection"], "method":wire["method"], "host":wire["host"], "target":wire["target"], "protocol":wire["protocol"]})
        })
        .collect();
    assert_eq!(
        json!(actual_requests),
        expected["native_http1_observations"]["requests"],
        "{name}: exact portable request sequence"
    );
    assert_eq!(
        close_idle_calls,
        expected["native_http1_observations"]["close_idle_calls"]
            .as_u64()
            .unwrap()
    );
    assert_eq!(
        json!(closures),
        expected["native_http1_observations"]["peer_closures_at_barriers"],
        "{name}: physical shutdown barriers"
    );
    assert_eq!(
        snapshot["peer_closed_before_cleanup"]
            .as_array()
            .unwrap()
            .len(),
        closures.len(),
        "{name}: unexpected extra physical shutdown"
    );
    let expected_public: Vec<Value> = expected["public_sdk"]
        .as_array()
        .unwrap()
        .iter()
        .map(|result| {
            let mut result = result.clone();
            result.as_object_mut().unwrap().remove("close_success");
            result
        })
        .collect();
    assert_eq!(public, expected_public, "{name}: public SDK outcomes");
    json!({"name":name, "public_sdk":public, "native_http1_observations":{"close_idle_calls":close_idle_calls, "requests":actual_requests, "peer_closures_at_barriers":closures}, "not_compared":["Go-private PutIdleConn callbacks and error strings", "Go Body.Close success"]})
}

async fn active_response(phase: &str, repeats: u64) {
    let name = format!("active-{phase}-retired-after-close");
    let mut peer = Peer::new(&name);
    let client = peer.owned_client();
    peer.plan("active", 42, phase, false);
    let task = artwork(&client, 42);
    peer.arrived("active", phase, 1);
    for _ in 0..repeats {
        client.close_idle_connections();
    }
    assert!(
        !task.is_finished(),
        "withheld response must still be active"
    );
    peer.release("active");
    let mut public = vec![finish_artwork(task, "active", 42).await];
    peer.closed(1);
    public.push(complete_artwork(&mut peer, &client, "next", 43, 2).await);
    let witness = compare_native(
        &mut peer,
        &name,
        public,
        repeats,
        vec![json!({"barrier":"active response completed", "connection":1})],
    );
    client.close_idle_connections();
    peer.closed(2);
    peer.finish(witness);
}

#[tokio::test(flavor = "multi_thread", worker_threads = 2)]
async fn active_headers_survive_close_then_the_real_tls_peer_retires() {
    active_response("headers", 1).await;
}

#[tokio::test(flavor = "multi_thread", worker_threads = 2)]
async fn active_body_survives_repeated_close_then_the_real_tls_peer_retires() {
    active_response("body", 2).await;
}

#[tokio::test(flavor = "multi_thread", worker_threads = 2)]
async fn new_acquisition_resets_future_idle_retirement_and_reuses_the_original_peer() {
    let name = "next-acquisition-resets-newly-idle-retirement";
    let mut peer = Peer::new(name);
    let client = peer.owned_client();
    peer.plan("active", 42, "body", false);
    let active = artwork(&client, 42);
    peer.arrived("active", "body", 1);
    client.close_idle_connections();
    peer.plan("resetting", 43, "headers", false);
    let resetting = artwork(&client, 43);
    peer.arrived("resetting", "headers", 2);
    peer.release("active");
    let active_result = finish_artwork(active, "active", 42).await;
    let reused_result = complete_artwork(&mut peer, &client, "reused", 44, 1).await;
    assert!(!resetting.is_finished());
    peer.release("resetting");
    let resetting_result = finish_artwork(resetting, "resetting", 43).await;
    let witness = compare_native(
        &mut peer,
        name,
        vec![active_result, resetting_result, reused_result],
        1,
        vec![],
    );
    client.close_idle_connections();
    peer.closed(1);
    peer.closed(2);
    peer.finish(witness);
}

#[tokio::test(flavor = "multi_thread", worker_threads = 2)]
async fn repeated_close_retires_idle_and_future_idle_without_interrupting_active_body() {
    let name = "mixed-idle-and-active-repeated-close";
    let mut peer = Peer::new(name);
    let client = peer.owned_client();
    let warm = complete_artwork(&mut peer, &client, "warm", 41, 1).await;
    peer.plan("active", 42, "body", false);
    let active = artwork(&client, 42);
    peer.arrived("active", "body", 1);
    let idle = complete_artwork(&mut peer, &client, "idle", 43, 2).await;
    client.close_idle_connections();
    client.close_idle_connections();
    peer.closed(2);
    assert!(!active.is_finished());
    peer.release("active");
    let active = finish_artwork(active, "active", 42).await;
    peer.closed(1);
    let next = complete_artwork(&mut peer, &client, "next", 44, 3).await;
    let witness = compare_native(
        &mut peer,
        name,
        vec![warm, active, idle, next],
        2,
        vec![
            json!({"barrier":"idle pool close", "connection":2}),
            json!({"barrier":"active response completed", "connection":1}),
        ],
    );
    client.close_idle_connections();
    peer.closed(3);
    peer.finish(witness);
}

async fn resource_reference(peer: &mut Peer, client: &Arc<NativeClient>) -> ResourceRef {
    peer.plan("metadata", 42, "", false);
    let client = Arc::clone(client);
    let task = tokio::spawn(async move { client.artwork_pages(42).await.unwrap() });
    peer.arrived("metadata", "", 1);
    let pages = tokio::time::timeout(BOUND, task).await.unwrap().unwrap();
    assert_eq!(pages.len(), 1);
    pages[0].image.resource.reference.clone()
}

async fn open_resource(
    peer: &mut Peer,
    client: &Arc<NativeClient>,
    reference: &ResourceRef,
    step: &str,
    phase: &str,
    connection: u64,
) -> pixiv_sdk::resource::ResourceResponse<pixiv_sdk::transport::ResourceBody> {
    peer.plan(step, 0, phase, true);
    let client = Arc::clone(client);
    let reference = reference.clone();
    let task = tokio::spawn(async move {
        client
            .open_resource(OpenResourceRequest {
                reference,
                ..Default::default()
            })
            .await
            .unwrap()
    });
    peer.arrived(step, phase, connection);
    let response = tokio::time::timeout(BOUND, task).await.unwrap().unwrap();
    assert_eq!(response.status_code, 200);
    response
}

#[tokio::test(flavor = "multi_thread", worker_threads = 2)]
async fn resource_eof_preserves_all_bytes_then_retires_the_active_media_peer() {
    let name = "resource-eof-after-close-retires-active-connection";
    let mut peer = Peer::new(name);
    let client = peer.owned_client();
    let reference = resource_reference(&mut peer, &client).await;
    let mut response =
        open_resource(&mut peer, &client, &reference, "active-stream", "body", 2).await;
    let mut first = [0];
    tokio::time::timeout(BOUND, response.body.read_exact(&mut first))
        .await
        .unwrap()
        .unwrap();
    assert_eq!(first, *b"a");
    client.close_idle_connections();
    peer.closed(1);
    peer.release("active-stream");
    let mut body = String::from("a");
    tokio::time::timeout(BOUND, response.body.read_to_string(&mut body))
        .await
        .unwrap()
        .unwrap();
    assert_eq!(body, "abc");
    drop(response);
    peer.closed(2);
    let mut public = vec![
        json!({"step":"metadata", "operation":"ArtworkPages", "success":true, "id":42}),
        json!({"step":"active-stream", "operation":"OpenResource", "success":true, "status":200, "body":body}),
    ];
    for step in ["next-stream", "reused-stream"] {
        let mut response = open_resource(&mut peer, &client, &reference, step, "", 3).await;
        let mut body = String::new();
        tokio::time::timeout(BOUND, response.body.read_to_string(&mut body))
            .await
            .unwrap()
            .unwrap();
        assert_eq!(body, "abc");
        public.push(json!({"step":step, "operation":"OpenResource", "success":true, "status":200, "body":body}));
    }
    let witness = compare_native(
        &mut peer,
        name,
        public,
        1,
        vec![
            json!({"barrier":"idle API pool close", "connection":1}),
            json!({"barrier":"stream EOF", "connection":2}),
        ],
    );
    client.close_idle_connections();
    peer.closed(3);
    peer.finish(witness);
}

#[tokio::test(flavor = "multi_thread", worker_threads = 2)]
async fn owned_and_caller_shared_native_pools_match_four_go_lifecycle_schedules() {
    let contract: Value =
        serde_json::from_str(include_str!("fixtures/client-ownership.json")).unwrap();
    let rows = contract["native_lifecycle"].as_array().unwrap();
    assert_eq!(rows.len(), 4);
    for expected in rows {
        let name = expected["name"].as_str().unwrap();
        let mut peer = Peer::new(name);
        let shared = peer.transport();
        let owns = expected["owned"].as_bool().unwrap();
        let pacing = Duration::from_millis(expected["pacing_ms"].as_u64().unwrap());
        let (a, b) = if owns {
            (
                Client::with_owned_transport("fixture-access", shared).unwrap(),
                Client::with_owned_transport("fixture-access", peer.transport()).unwrap(),
            )
        } else {
            (
                Client::try_with_transport("fixture-access", shared.clone()).unwrap(),
                Client::try_with_transport("fixture-access", shared).unwrap(),
            )
        };
        let a = Arc::new(a.with_pacing(pacing));
        let b = Arc::new(b.with_pacing(pacing));
        let expected_connections: Vec<u64> = expected["requests"]
            .as_array()
            .unwrap()
            .iter()
            .map(|wire| wire["connection"].as_u64().unwrap())
            .collect();
        for (index, client) in [&a, &a, &b, &b].into_iter().enumerate() {
            complete_artwork(
                &mut peer,
                client,
                &format!("request-{index}"),
                73,
                expected_connections[index],
            )
            .await;
        }
        a.close_idle_connections();
        a.close_idle_connections();
        if owns {
            peer.closed(1);
        }
        for (index, client) in [(4, &a), (5, &b)] {
            complete_artwork(
                &mut peer,
                client,
                &format!("request-{index}"),
                73,
                expected_connections[index],
            )
            .await;
        }
        let snapshot = peer.snapshot();
        let requests: Vec<Value> = snapshot["requests"]
            .as_array()
            .unwrap()
            .iter()
            .map(|wire| {
                json!({"connection":wire["connection"], "method":wire["method"], "host":wire["host"], "path":wire["target"].as_str().unwrap().split('?').next().unwrap(), "authorization":wire["authorization"], "user_id":wire["user_id"], "accept_language":wire["accept_language"], "refresh_token":wire["refresh_token"]})
            })
            .collect();
        assert_eq!(json!(requests), expected["requests"], "{name}");
        assert_eq!(
            snapshot["peer_closed_before_cleanup"]
                .as_array()
                .unwrap()
                .len() as u64,
            expected["connections_closed_before_test_cleanup"]
                .as_u64()
                .unwrap(),
            "{name}"
        );
        let witness = json!({"name":name, "owned":owns, "pacing_ms":expected["pacing_ms"], "requests":requests, "connections_closed_before_test_cleanup":snapshot["peer_closed_before_cleanup"].as_array().unwrap().len(), "route":"normal SDK ownership transfer or caller-owned injection of a custom native reqwest client"});
        a.close_idle_connections();
        b.close_idle_connections();
        if owns {
            peer.closed(2);
            peer.closed(3);
        }
        peer.finish(witness);
    }
}
