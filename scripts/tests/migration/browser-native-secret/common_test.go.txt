package chromium

import (
	"context"
	"encoding/hex"
	"encoding/json"
	"os"
	"testing"
)

func secretHex(t *testing.T, value string) []byte {
	t.Helper()
	body, err := hex.DecodeString(value)
	if err != nil {
		t.Fatal(err)
	}
	return body
}

func secretError(err error) string {
	if err == nil {
		return ""
	}
	return err.Error()
}

func secretResult(t *testing.T, value any) {
	t.Helper()
	body, err := json.MarshalIndent(value, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(os.Getenv("PIXIV_NATIVE_RESULT"), append(body, '\n'), 0o600); err != nil {
		t.Fatal(err)
	}
}

type checkedContext struct {
	context.Context
	checks      int
	cancel      string
	cancelCheck int
	trigger     context.CancelFunc
}

func (c *checkedContext) Err() error {
	c.checks++
	check := c.cancelCheck
	if check == 0 {
		check = 2
	}
	if c.cancel == "after-copy" && c.checks >= check {
		c.trigger()
	}
	return c.Context.Err()
}
